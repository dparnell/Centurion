// DCX counts a 16 bit register down to zero and BNZ leaves the loop when it
// gets there. Three stars means DCX set the zero flag on exactly the right
// iteration.
//
// The leading NOPs are load bearing. The reset vector is JMP 0x8001, so the
// byte at 0x8000 is never executed and a program placed there starts one byte
// in. With the LDXW at 0x8000 the machine began at its first operand byte,
// which cost a whole round of "DCX is broken for 16 bit values" - it is not.
// One filler byte is enough; with it, literals 0x0101, 0x0180 and 0x02ff count
// out to exactly 257, 384 and 767. hellorld.txt and bnz_test.txt open with
// NOPs for the same reason.
01 // NOP
01 // NOP
60 // LDXW #0003
00
03
80 // LDAL '*'
2a
a1 // STAL #f201  (print)
f2
01
3f // DCX
15 // BNZ -6, back to the store
fa
80 // LDAL '\n'
0a
a1 // STAL #f201
f2
01
80 // LDAL #01
01
a1 // STAL #f900  (sim_end)
f9
00
